/** "/a/b/" and "a/b" both become "/a/b"; "" and "/" are the root. */
export function normalizePath(p: string): string {
  const parts = p.split("/").map((x) => x.trim()).filter((x) => x && x !== "." && x !== "..");
  return `/${parts.join("/")}`;
}
